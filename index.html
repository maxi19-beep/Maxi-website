import pygame import sys pygame.init() screen = pygame.display.set_mode((400, 600)) clock = pygame.time.Clock() # Colors and Road variables BG_COLOR = (0, 0, 225) # Grass green background ROAD_COLOR = (255, 0, 0) # Dark asphalt gray LINE_COLOR = (0, 0, 0) # White lane markings road_scroll = 9 # Keeps track of road movement # Main game loop while True: for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() # Scroll the road downward to create motion road_scroll = (road_scroll + 5) % 40 # Draw background and road screen.fill(BG_COLOR) pygame.draw.rect(screen, ROAD_COLOR, (100, 0, 200, 600)) # Draw moving white center dashed lane lines for i in range(-40, 640, 40): pygame.draw.rect(screen, LINE_COLOR, (195, i + road_scroll, 10, 20)) # Refresh screen and cap frame rate pygame.display.flip() clock.tick(60)
